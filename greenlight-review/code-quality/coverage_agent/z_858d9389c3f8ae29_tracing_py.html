2OpenTelemetry Tracing for Agent Service.

7import os 

8from typing import Optional 

9 

10# Check if OpenTelemetry is available 

11OTEL_AVAILABLE = False 

12 

13try: 

14 from opentelemetry import trace 

15 from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter 

16 from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor 

17 from opentelemetry.instrumentation.requests import RequestsInstrumentor 

18 from opentelemetry.sdk.resources import Resource 

19 from opentelemetry.sdk.trace import TracerProvider 

20 from opentelemetry.sdk.trace.export import BatchSpanProcessor 

21 from opentelemetry.trace.propagation.tracecontext import TraceContextTextMapPropagator 

22 from opentelemetry.propagate import set_global_textmap 

23 

24 OTEL_AVAILABLE = True 

25except ImportError: 

26 trace = None

29def setup_tracing(app, service_name: str = "agent") -> bool: 

30 """ 

31 Setup OpenTelemetry tracing for the Agent FastAPI application. 

32  

33 Args: 

34 app: FastAPI application instance 

35 service_name: Name of the service for traces 

36  

37 Returns: 

38 True if tracing was configured, False otherwise 

39 """ 

40 if not OTEL_AVAILABLE: 

41 print("OpenTelemetry not available. Install with: pip install opentelemetry-api opentelemetry-sdk opentelemetry-instrumentation-fastapi opentelemetry-exporter-otlp-proto-grpc") 

42 return False 

43 

44 # Check if tracing is enabled 

45 otel_enabled = os.getenv("OTEL_ENABLED", "false").lower() == "true" 

46 if not otel_enabled: 

47 print("OpenTelemetry tracing is disabled (OTEL_ENABLED != true)") 

48 return False 

49 

50 # Get configuration from environment 

51 service_name = os.getenv("OTEL_SERVICE_NAME", service_name) 

52 otlp_endpoint = os.getenv("OTEL_EXPORTER_OTLP_ENDPOINT", "http://signoz-otel-collector:4317") 

53 

54 try: 

55 # Create resource with service information 

56 resource = Resource.create({ 

57 "service.name": service_name, 

58 "service.version": "1.0.0", 

59 "deployment.environment": os.getenv("ENVIRONMENT", "development"), 

60 }) 

61 

62 # Create tracer provider 

63 provider = TracerProvider(resource=resource) 

64 

65 # Configure OTLP exporter 

66 exporter = OTLPSpanExporter(endpoint=otlp_endpoint, insecure=True) 

67 provider.add_span_processor(BatchSpanProcessor(exporter)) 

68 

69 # Set the tracer provider 

70 trace.set_tracer_provider(provider) 

71 

72 # Set up W3C Trace Context propagator 

73 set_global_textmap(TraceContextTextMapPropagator()) 

74 

75 # Instrument FastAPI 

76 FastAPIInstrumentor.instrument_app(app) 

77 

78 # Instrument requests library for outgoing HTTP calls 

79 try: 

80 RequestsInstrumentor().instrument() 

81 except Exception as e: 

82 print(f"Failed to instrument requests library: {e}") 

83 

84 print(f"OpenTelemetry tracing configured for service: {service_name} -> {otlp_endpoint}") 

85 return True 

86 

87 except Exception as e: 

88 print(f"Failed to configure OpenTelemetry tracing: {e}") 

89 return False

92def get_tracer(name: str = "agent"): 

93 """Get a tracer instance.""" 

94 if not OTEL_AVAILABLE: 

95 return None
